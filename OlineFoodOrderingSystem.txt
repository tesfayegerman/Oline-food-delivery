// Import classes for handling date and time.
// LocalDateTime allows working with date and time without time zone.
// DateTimeFormatter is used to format and parse date-time objects into strings and vice versa.
import java.util.*; // Import all classes from the java.util package for collections and Scanner
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;

// Abstract base class representing a general user
abstract class User {
    String username, password, phone, email, address; // User's personal info fields

    // Constructor to initialize user details
    public User(String u, String p, String ph, String e, String a) {
        username = u; 
        password = p; 
        phone = ph; 
        email = e; 
        address = a;
    }

    // Abstract method that each subclass will implement to show menu options
    abstract void displayMenu();
}

// Customer class that extends User
class Customer extends User {
    Cart cart = new Cart(); // Each customer has their own shopping cart

    // Constructor to initialize a customer with user details
    public Customer(String u, String p, String ph, String e, String a) {
        super(u, p, ph, e, a); // Call parent constructor
    }

    // Implementation of displayMenu specifically for customers
    void displayMenu() {
        System.out.println("1. Browse Food\n2. Browse Drinks\n3. View Cart\n4. Checkout\n5. Logout");
    }
}

// Admin class that extends User
class Admin extends User {
    // Constructor to initialize an admin with user details
    public Admin(String u, String p, String ph, String e, String a) {
        super(u, p, ph, e, a); // Call parent constructor
    }

    // Implementation of displayMenu specifically for administrators
    void displayMenu() {
        System.out.println("1. Add Item\n2. Update Item\n3. Remove Item\n4. View Orders\n5. Logout");
    }
}

// Class representing a food or drink item
class FoodItem {
    int id; // Unique identifier for the menu item
    String name, category; // Name and category (Food/Drink)
    double price; // Price of the item

    // Constructor to initialize food item details
    public FoodItem(int i, String n, double p, String c) {
        id = i;
        name = n;
        price = p;
        category = c;
    }

    // Override toString method to display food item details
    public String toString() {
        return id + ". " + name + " - " + price + " ETB (" + category + ")";
    }
}

// Class representing a customer's shopping cart
class Cart {
    List<FoodItem> items = new ArrayList<>(); // List of items in the cart

    // Method to add an item to the cart
    void addItem(FoodItem item) {
        items.add(item);
        System.out.println(item.name + " added to cart.");
    }

    // Method to view all items in the cart
    void viewCart() {
        if (items.isEmpty()) { // Check if cart is empty
            System.out.println("Your cart is empty."); return; // Inform user if empty
        }
        double total = 0; // Initialize total cost
        System.out.println("Your Cart:");
        for (FoodItem item : items) {
            System.out.println(item); // Print item details
            total += item.price; // Sum up prices
        }
        System.out.println("Total: " + total + " ETB"); // Show total price
    }

    // Method to check if cart is empty
    boolean isEmpty() {
        return items.isEmpty();
    }

    // Method to perform checkout, returning total amount and clearing cart
    double checkout() {
        double total = items.stream().mapToDouble(i -> i.price).sum(); // Sum prices
        items.clear(); // Clear items after checkout
        return total; // Return total amount
    }
}

// Define a class representing a customer order
class Order {
    static int counter = 1; // Static variable to keep track of the total number of orders, used to generate unique order IDs

    int orderId;             // Unique identifier for each order
    String customer;         // Name of the customer who placed the order
    String status = "Pending"; // Current status of the order, defaulting to "Pending"
    double amount;           // Total amount for the order

    String orderDate;        // String to store formatted date and time when the order was placed
    String deliveryTime;     // String to store estimated delivery time

    // Constructor to initialize a new order with customer name and amount
    public Order(String c, double a) {
        orderId = counter++; // Assign current counter value as order ID and then increment counter

        customer = c;        // Set the customer name
        amount = a;          // Set the order amount

        // Capture the current date and time
        LocalDateTime now = LocalDateTime.now();

        // Define a formatter for date and time in "yyyy-MM-dd HH:mm" format
        DateTimeFormatter dtf = DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm");

        // Format current date and time and assign to orderDate
        orderDate = now.format(dtf);

        // Calculate estimated delivery time, set to 1 hour from now, and format it
        deliveryTime = now.plusHours(1).format(dtf);
    }

    // Method to represent the order details as a string
    public String toString() {
        return "Order#" + orderId + " by " + customer +
               " - " + amount + " ETB - Status: " + status +
               "\n  Ordered at: " + orderDate +
               "\n  Estimated Delivery: " + deliveryTime;
    }
}

                return; // Exit login method once logged in

}

             else if (ch == 3) {
